# Optional: SQL Server article source
# ─────────────────────────────────────────────────────────────────────────────
# Only needed to ingest articles from a SQL Server database (see
# suni/ingestion/articles.py and the SUNIVERSE_DB_* variables in .env.example).
# SUNI runs fine without this — article ingestion simply reports that it is not
# configured, and everything else is unaffected.
#
# The wheel installs on every platform, but `import pyodbc` needs the unixODBC
# RUNTIME library, not just the Python package. On Linux install that first:
#
#   Debian/Ubuntu:  sudo apt install unixodbc
#   RHEL/Fedora:    sudo dnf install unixODBC
#   macOS:          brew install unixodbc
#
# You also need a driver for your database, e.g. Microsoft's
# "ODBC Driver 17 for SQL Server", and SUNIVERSE_DB_DRIVER set to match.
#
#   pip install -r requirements-articles.txt

pyodbc==5.3.0
